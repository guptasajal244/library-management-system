# Open MySQL and type in the following.

create database library;

use library;


CREATE TABLE user (
    id INTEGER PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(200) NOT NULL,
    username VARCHAR(200) UNIQUE NOT NULL,
    password VARCHAR(200) NOT NULL,
    issued_books INTEGER DEFAULT 0
);

CREATE TABLE admin (
    id INTEGER PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(200) UNIQUE NOT NULL,
    password VARCHAR(200) NOT NULL
);

CREATE TABLE book (
    id INTEGER PRIMARY KEY AUTO_INCREMENT,
    isbn VARCHAR(50) UNIQUE NOT NULL,
    title VARCHAR(200) NOT NULL,
    author VARCHAR(200) NOT NULL,
    genre VARCHAR(100) NOT NULL,
    quantity INTEGER NOT NULL
);

CREATE TABLE issued_books (
    id INTEGER PRIMARY KEY AUTO_INCREMENT,
    isbn VARCHAR(50) NOT NULL,
    title VARCHAR(200) NOT NULL,
    user_id INTEGER NOT NULL,
    issued_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    return_date DATE
);

CREATE TABLE orders (
    id INTEGER PRIMARY KEY AUTO_INCREMENT,
    userId INTEGER NOT NULL,
    BookName VARCHAR(200) NOT NULL,
    Author VARCHAR(200) NOT NULL,
    orderDate DATE NOT NULL
);

# Insert the admin account.
# IMPORTANT: The password column must store a werkzeug hash, NOT plaintext.
# Storing plaintext will permanently break login because the app uses
# check_password_hash(), which only accepts werkzeug-format hashes.
#
# Generate the correct hash first (run in your terminal):
#   python -c "from werkzeug.security import generate_password_hash; print(generate_password_hash('your_password'))"
#
# Then paste the output into the INSERT below:
#
# INSERT INTO admin (username, password)
# VALUES ('admin', '<paste_hash_here>');
#
# Alternatively, run this one-liner which does it all automatically:
#   python -c "import pymysql; from werkzeug.security import generate_password_hash; c=pymysql.connect(host='localhost',user='root',password='root',database='library'); cur=c.cursor(); cur.execute('INSERT INTO admin (username, password) VALUES (%s, %s)', ('admin', generate_password_hash('admin'))); c.commit(); c.close(); print('Admin created.')"
